# AI Core 3D — Fundação + Reator — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Substituir o reator do hero por um objeto Three.js real, com volume, luz e oclusão, dirigido pelo mesmo cérebro (`EnergyField.core`) que já existe.

**Architecture:** Three.js entra como ESM vendorado, carregado por `<script type="module">`, sem bundler. `main.js` continua a IIFE monolítica e publica uma ponte mínima (`window.__beyon3d`). A camada 3D vive em `js3d/`, com dependência apontando só para dentro: `reactor → materials → stage → three`. O tick do reator é registrado em `EnergyField.add()`, então continua havendo **um rAF só**.

**Tech Stack:** Three.js 0.185.1 (ESM), HTML/CSS/JS puro, sem build. Testes por página de harness carregada no navegador.

## Global Constraints

- **Sem build.** Nenhum bundler, nenhum passo de compilação, nenhum `node_modules` em runtime.
- **CSP `script-src 'self'`.** Nada inline, nada de CDN, nada de import map.
- **Um rAF só.** Nunca `renderer.setAnimationLoop`. O tick é registrado via `EnergyField.add(tick)`.
- **Rampa sagrada, imutável em repouso.** `--void #000000` · `--deep #06272B` · `--mid #0F6A73` · `--glow #00F5FF` · `--hi #9FF8FF` · `--core #F2FEFF`. Dourado só no Estado Singular, e vindo de `plasmaColor`.
- **`watchQuality` é uma via.** 1 → 0.62 → 0.38, piso 0.38, nunca sobe. Ao gatear: reduza QUANTIDADE, nunca presença.
- **`dpr = Math.min(devicePixelRatio, 1.75)`** — constante de nascença, não degrau de degradação.
- **Sem pós-processamento.** Nada de `EffectComposer`, `UnrealBloomPass`, `BokehPass`. Brilho vem de geometria aditiva.
- **O reator canvas 2D (`heroReactor`) NÃO é removido neste plano.** Ele fica no código, apenas escondido. A remoção é uma etapa posterior.

```js
/* Micro-harness. Sem dependências, sem runner. Publica em window.__testResults. */
const results = [];

export function check(name, fn) {
  try {
    const r = fn();
    if (r === true) results.push({ name, ok: true });
    else results.push({ name, ok: false, msg: String(r) });
  } catch (e) {
    results.push({ name, ok: false, msg: e && e.message ? e.message : String(e) });
  }
}

export function near(actual, expected, tol) {
  return Math.abs(actual - expected) <= tol
    ? true
    : `esperado ${expected} ±${tol}, veio ${actual}`;
}

export function close() {
  window.__testResults = results;
  const failed = results.filter((r) => !r.ok);
  window.__testSummary = { total: results.length, failed: failed.length };
  const el = document.getElementById('out');
  if (el) {
    el.textContent = results
      .map((r) => (r.ok ? 'PASS  ' : 'FAIL  ') + r.name + (r.msg ? '  → ' + r.msg : ''))
      .join('\n') + `\n\n${results.length - failed.length}/${results.length} passaram`;
  }
}
```

```js
/* STAGE — renderer, cena, câmera e registro de objetos.
   Deliberadamente agnóstico: não sabe o que é "hero" nem o que é "reator".
   É isso que permite, no sub-projeto 5 (cena com scroll), trocar só o
   tamanho e a câmera sem tocar nos objetos. */
import * as THREE from '../vendor/three.module.min.js';

const DPR_CAP = 1.75;   /* constante de nascença, não degrau de degradação */
const FOV = 35;         /* volume sem distorção de lente */
const CAM_Z = 6.2;

export function createStage(canvas) {
  let renderer;
  try {
    renderer = new THREE.WebGLRenderer({
      canvas,
      alpha: true,
      antialias: true,
      powerPreference: 'high-performance'
    });
  } catch (e) {
    return null;                       /* sem WebGL: quem chamou cai no fallback */
  }
  if (!renderer || !renderer.getContext()) return null;

  renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, DPR_CAP));
  renderer.setClearColor(0x000000, 0);

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(FOV, 1, 0.1, 100);
  camera.position.set(0, 0, CAM_Z);
  camera.lookAt(0, 0, 0);

  const registry = { nodes: {}, meshes: {}, materials: {} };

  /* Como quase tudo no reator é emissivo/aditivo, é preciso pouca luz: uma
     direcional que dá FORMA à esfera (senão o MeshStandardMaterial some) e uma
     hemisférica de preenchimento. Cor branca de propósito: a cor do objeto vem
     dos materiais, que saem de plasmaColor. Luz colorida introduziria um
     segundo croma pela porta dos fundos. */
  const key = new THREE.DirectionalLight(0xffffff, 1.15);
  key.position.set(2.4, 3.0, 3.2);
  scene.add(key);

  const fill = new THREE.HemisphereLight(0xffffff, 0x0b2f35, 0.55);
  scene.add(fill);

  const stage = {
    scene,
    camera,
    renderer,
    registry,
    lights: { key, fill },
    quality: 1,

    /* Todo objeto entra por aqui e fica endereçável pelo nome — é o que dá
       alvos prontos de GSAP e o que torna o sub-projeto 3 (peças clicáveis)
       quase de graça. */
    add(name, obj, parentName) {
      const parent = (parentName && registry.nodes[parentName]) || scene;
      parent.add(obj);
      registry.nodes[name] = obj;
      if (obj.isMesh) registry.meshes[name] = obj;
      return obj;
    },

    /* watchQuality é UMA VIA (1 → 0.62 → 0.38). Cada degrau reduz
       QUANTIDADE, nunca presença: o objeto continua inteiro. */
    setQuality(q) {
      stage.quality = q;
      const full = q >= 0.999;
      renderer.shadowMap.enabled = full;
      scene.environment = full ? stage._env || null : null;
    },

    setEnvironment(tex) {
      stage._env = tex;
      if (stage.quality >= 0.999) scene.environment = tex;
    },

    resize(w, h) {
      if (!w || !h) return;
      renderer.setSize(w, h, false);
      camera.aspect = w / h;
      camera.updateProjectionMatrix();
    },

    render() {
      renderer.render(scene, camera);
    },

    dispose() {
      renderer.dispose();
    }
  };

  stage.setQuality(1);
  stage.resize(canvas.clientWidth || canvas.width, canvas.clientHeight || canvas.height);
  return stage;
}
```

- [ ] **Step 3: Implementar `js3d/materials.js`**

```js
/* MATERIAIS — a cor NUNCA é lida do CSS por frame (getComputedStyle em loop
   é veneno). Ela sai de plasmaColor(warmth, role), a mesma função que rege o
   resto do site. Consequência: a metamorfose do Estado Singular arrasta o
   objeto 3D para o dourado de graça e em sincronia garantida com CSS,
   tipografia e partículas — porque é literalmente a mesma fonte.

   Linguagem: técnico-emissivo, não PBR realista. Metal ou vidro brigaria com
   a lei do projeto e custaria caro; tudo aditivo sobre preto reproduz a
   estética atual sem introduzir um segundo croma. */
import * as THREE from '../vendor/three.module.min.js';

const GLOW = 0, HI = 1, CORE = 2;

function applyTriple(color, triple) {
  const p = triple.split(',');
  color.setRGB(+p[0] / 255, +p[1] / 255, +p[2] / 255);
}

export function createMaterials(plasmaColor) {
  const shell = new THREE.MeshStandardMaterial({
    color: 0x061a1e, roughness: 0.42, metalness: 0.12, emissiveIntensity: 1
  });
  shell.emissive = new THREE.Color();

  const rim = new THREE.MeshBasicMaterial({
    transparent: true, opacity: 0.34,
    blending: THREE.AdditiveBlending, side: THREE.BackSide, depthWrite: false
  });

  const wire = new THREE.MeshBasicMaterial({
    wireframe: true, transparent: true, opacity: 0.16,
    blending: THREE.AdditiveBlending, depthWrite: false
  });

  const edge = new THREE.LineBasicMaterial({
    transparent: true, opacity: 0.72,
    blending: THREE.AdditiveBlending, depthWrite: false
  });

  const orbit = new THREE.MeshBasicMaterial({
    transparent: true, opacity: 0.30,
    blending: THREE.AdditiveBlending, depthWrite: false
  });

  const plasma = new THREE.MeshBasicMaterial({
    transparent: true, opacity: 0.95,
    blending: THREE.AdditiveBlending, depthWrite: false
  });

  const node = new THREE.MeshBasicMaterial({
    transparent: true, opacity: 0.85,
    blending: THREE.AdditiveBlending, depthWrite: false
  });

  const mats = {
    shell, rim, wire, edge, orbit, plasma, node,
    _warmth: -1,
    _writes: 0,

    /* Gated por mudança real, mesmo padrão do setSingCss no main.js. */
    update(warmth) {
      if (Math.abs(warmth - mats._warmth) < 0.002) return;
      mats._warmth = warmth;
      mats._writes += 1;
      const g = plasmaColor(warmth, GLOW);
      const h = plasmaColor(warmth, HI);
      const c = plasmaColor(warmth, CORE);
      applyTriple(rim.color, g);
      applyTriple(wire.color, g);
      applyTriple(orbit.color, g);
      applyTriple(edge.color, h);
      applyTriple(node.color, h);
      applyTriple(plasma.color, c);
      applyTriple(shell.emissive, g);
    },

    dispose() {
      [shell, rim, wire, edge, orbit, plasma, node].forEach((m) => m.dispose());
    }
  };

  mats.update(0);
  return mats;
}
```

```js
/* BOOT — o único módulo que conhece o DOM e o hero.
   Sequência obrigatória: qualquer falha desiste em silêncio e o SVG estático
   permanece visível. Nunca lança: o resto do site não pode nem ficar sabendo. */
import { createStage } from './stage.js';
import { createMaterials } from './materials.js';
import { createReactor } from './reactor.js';

const SEL_CANVAS = '.hero__reactor3d';
const SEL_FALLBACK = '.hero__reactor-fallback';

export function boot(doc, bridge) {
  if (!bridge || !bridge.field || typeof bridge.plasmaColor !== 'function') {
    return { ok: false, reason: 'sem-ponte' };
  }
  const canvas = doc.querySelector(SEL_CANVAS);
  if (!canvas) return { ok: false, reason: 'sem-canvas' };

  const stage = createStage(canvas);
  if (!stage) return { ok: false, reason: 'sem-webgl' };

  let mats, reactor;
  try {
    mats = createMaterials(bridge.plasmaColor);
    reactor = createReactor(stage, mats);
  } catch (e) {
    stage.dispose();
    return { ok: false, reason: 'erro-montagem' };
  }

  const fallback = doc.querySelector(SEL_FALLBACK);
  if (fallback && fallback.style) fallback.style.display = 'none';

  const field = bridge.field;
  let last = 0;
  let lastQuality = -1;

  function measure() {
    const w = canvas.clientWidth || canvas.width;
    const h = canvas.clientHeight || canvas.height;
    stage.resize(w, h);
    /* ÂNCORA: o campo de partículas do site orbita o reator. Sem isso as
       partículas voltariam ao centro da tela. */
    const r = canvas.getBoundingClientRect
      ? canvas.getBoundingClientRect()
      : { left: 0, top: 0, width: w, height: h };
    field.coreAnchor = { x: r.left + r.width / 2, y: r.top + r.height / 2 };
  }

  function tick(now) {
    if (typeof document !== 'undefined' && document.hidden) return;
    const dt = last ? Math.min(60, now - last) : 16;
    last = now;

    const q = field.quality == null ? 1 : field.quality;
    if (q !== lastQuality) {
      lastQuality = q;
      stage.setQuality(q);
      reactor.setQuality(q);
    }

    measure();
    const core = field.core || {};
    const warmth = (field.sing && field.sing.warmth) || 0;
    reactor.tick(core, field.E == null ? 0 : field.E, warmth, dt);
    stage.render();
  }

  measure();

  if (bridge.reduced) {
    reactor.tick(field.core || {}, 0, 0, 16);
    stage.render();                       /* presença sem movimento */
    return { ok: true, stage, reactor, tick };
  }

  field.add(tick);                        /* UM rAF só: o loop do organismo */
  return { ok: true, stage, reactor, tick };
}
```

```js
/* Mede o matiz medio dos pixels acesos do framebuffer.
   Ciano ≈ 183°, dourado plasma ≈ 50°. */
function medirCena(stageAlvo, reactorAlvo, warmth) {
  const gl = stageAlvo.renderer.getContext();
  const w = gl.drawingBufferWidth, h = gl.drawingBufferHeight;
  for (let i = 0; i < 20; i += 1) reactorAlvo.tick(coreStub, 0.8, warmth, 16);
  stageAlvo.render();
  const buf = new Uint8Array(w * h * 4);
  gl.readPixels(0, 0, w, h, gl.RGBA, gl.UNSIGNED_BYTE, buf);
  let acesos = 0, somaSin = 0, somaCos = 0;
  for (let p = 0; p < buf.length; p += 4) {
    const r = buf[p] / 255, g = buf[p + 1] / 255, b = buf[p + 2] / 255;
    const max = Math.max(r, g, b), min = Math.min(r, g, b);
    if (max < 0.10) continue;
    acesos += 1;
    const d = max - min;
    let hue = 0;
    if (d > 0.0001) {
      if (max === r) hue = 60 * (((g - b) / d) % 6);
      else if (max === g) hue = 60 * ((b - r) / d + 2);
      else hue = 60 * ((r - g) / d + 4);
    }
    if (hue < 0) hue += 360;
    const rad = hue * Math.PI / 180;
    somaSin += Math.sin(rad); somaCos += Math.cos(rad);
  }
  let matiz = Math.atan2(somaSin / (acesos || 1), somaCos / (acesos || 1)) * 180 / Math.PI;
  if (matiz < 0) matiz += 360;
  return { acesos, matiz, total: w * h };
}
```
